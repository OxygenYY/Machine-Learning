# 一、Hugging Face Transformers 快速上手

Hugging Face 的 Transformers 库提供了统一的接口来加载和使用各类预训练模型。核心入口是 `AutoModel` 和 `AutoTokenizer`：

```python
from transformers import (
    AutoModelForCausalLM,
    AutoTokenizer,
    pipeline
)
import torch

# ---- 加载模型和分词器 ----
model_name = "gpt2"  # 可替换为 "meta-llama/Llama-2-7b-hf" / "mistralai/Mistral-7B-v0.1"

tokenizer = AutoTokenizer.from_pretrained(model_name)

# 设置 pad_token（GPT-2 没有 pad token，用 eos 代替）
if tokenizer.pad_token is None:
    tokenizer.pad_token = tokenizer.eos_token

# 加载模型
model = AutoModelForCausalLM.from_pretrained(
    model_name,
    torch_dtype=torch.float16,      # 半精度加载节省显存
    device_map="auto",               # 自动分配到可用设备
    # load_in_8bit=True,             # 8-bit 量化（需要 bitsandbytes）
)

print(f"模型参数量: {sum(p.numel() for p in model.parameters()):,}")
print(f"词表大小: {model.config.vocab_size}")
```

# 三、推理与生成

使用 `model.generate()` 进行自回归文本生成。理解各个采样参数至关重要：

```python
def generate_text(model, tokenizer, prompt, **gen_kwargs):
    """封装生成逻辑"""
    inputs = tokenizer(prompt, return_tensors="pt").to(model.device)

    with torch.no_grad():
        outputs = model.generate(
            inputs.input_ids,
            attention_mask=inputs.attention_mask,
            pad_token_id=tokenizer.pad_token_id,
            eos_token_id=tokenizer.eos_token_id,
            **gen_kwargs
        )

    generated = tokenizer.decode(outputs[0], skip_special_tokens=True)
    return generated


# ---- 关键采样参数 ----
prompt = "The future of artificial intelligence is"

# (a) 贪婪解码：每步选概率最高的 token，确定性强但可能重复
output_greedy = generate_text(
    model, tokenizer, prompt,
    max_new_tokens=50,
    do_sample=False,
    temperature=1.0,
)
print("【贪婪解码】\n", output_greedy, "\n")

# (b) 带温度的随机采样：temperature < 1 → 更集中（确定），> 1 → 更平坦（随机）
output_temp = generate_text(
    model, tokenizer, prompt,
    max_new_tokens=50,
    do_sample=True,
    temperature=0.7,    # 常用值: 0.6~0.9
    top_k=50,           # 只从概率最高的 50 个 token 中采样
    top_p=0.9,          # 累积概率超过 0.9 前的最小集合
    repetition_penalty=1.1,  # 惩罚重复
)
print("【随机采样 (t=0.7, top-k=50, top-p=0.9)】\n", output_temp)
```

参数速查表：

| 参数 | 作用 | 典型值 |
|------|------|--------|
| `temperature` | 控制分布"锐度"，低→确定，高→随机 | 0.6~0.9 |
| `top_k` | 只从 Top-K 个 token 中采样 | 40~100 |
| `top_p` | 累积概率截断（核采样） | 0.9~0.95 |
| `repetition_penalty` | 惩罚已出现的 token | 1.0~1.2 |
| `max_new_tokens` | 最大生成长度 | 100~4096 |

# 五、LoRA：低秩适配

**LoRA（Low-Rank Adaptation）** 是当前最流行的参数高效微调（PEFT）方法。核心思想：权重矩阵的更新 $\Delta W$ 通常是低秩的，可以分解为两个小矩阵的乘积：

$$
W' = W + \Delta W = W + BA
$$

其中 $W \in \mathbb{R}^{d \times k}$（冻结），$B \in \mathbb{R}^{d \times r}$，$A \in \mathbb{R}^{r \times k}$，且 $r \ll \min(d, k)$。训练时只更新 $A$ 和 $B$：

```python
import torch
import torch.nn as nn

class LoRALayer(nn.Module):
    """手动实现 LoRA 层的核心逻辑"""
    def __init__(self, in_features, out_features, rank=8, alpha=16, dropout=0.0):
        super().__init__()
        self.rank = rank
        self.alpha = alpha
        self.scaling = alpha / rank  # 缩放因子

        # 原始权重矩阵（冻结）
        self.weight = nn.Parameter(torch.randn(out_features, in_features) * 0.02)
        self.weight.requires_grad = False

        # LoRA 低秩矩阵（可训练）
        # A: (in_features, rank)，用 Kaiming 初始化
        self.lora_A = nn.Parameter(torch.zeros(in_features, rank))
        # B: (rank, out_features)，初始化为零
        self.lora_B = nn.Parameter(torch.zeros(rank, out_features))

        nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
        # B 保持零初始化，确保训练开始时 ∆W = 0

        self.dropout = nn.Dropout(dropout) if dropout > 0 else nn.Identity()

    def forward(self, x):
        # 原始前向传播（冻结）
        original = F.linear(x, self.weight)

        # LoRA 前向传播（训练）
        x_dropped = self.dropout(x)
        lora_out = (x_dropped @ self.lora_A @ self.lora_B) * self.scaling

        return original + lora_out

# 测试 LoRA
batch, in_dim, out_dim = 4, 128, 256
x = torch.randn(batch, in_dim)
lora_linear = LoRALayer(in_dim, out_dim, rank=4, alpha=8)
out = lora_linear(x)

# 参数比较
original_params = in_dim * out_dim
lora_trainable = in_dim * 4 + 4 * out_dim  # A + B

print(f"LoRA 可训练参数: {lora_trainable:,}")
print(f"参数减少: {(1 - lora_trainable/original_params)*100:.1f}%")
```

# 八、量化技术深度解析

量化将高精度浮点数（FP32/FP16）映射到低位宽整数，大幅降低显存占用：

```python
import torch
import numpy as np

# ---- 量化原理演示 ----
def absmax_quantize(tensor, bits=8):
    """绝对最大值量化"""
    qmax = 2 ** (bits - 1) - 1  # e.g., 127 for int8
    scale = tensor.abs().max() / qmax
    quantized = torch.round(tensor / scale).clamp(-qmax, qmax).to(torch.int8)
    return quantized, scale

def absmax_dequantize(quantized, scale):
    """反量化"""
    return quantized.float() * scale

# 测试
x = torch.tensor([0.1, -0.5, 0.8, -1.2, 0.01, 2.5])
x_q, scale = absmax_quantize(x, bits=8)
x_dq = absmax_dequantize(x_q, scale)

print(f"原始张量: {x}")
print(f"量化后(int8): {x_q}")
print(f"反量化: {x_dq}")
print(f"量化误差: {(x - x_dq).abs().mean():.6f}")
```

# 十、最佳实践总结

1. **选择合适的基础模型**：中文场景优先选 Qwen、ChatGLM；代码场景选 DeepSeek-Coder；通用场景选 Llama、Mistral
2. **数据质量远重于数据数量**：SFT 阶段 1000 条高质量数据 > 10000 条低质量数据
3. **使用 ChatML / ShareGPT 模板**：确保训练数据使用与预训练模型一致的对话格式
4. **监控训练损失和评估损失**：验证集损失开始上升时立即停止，避免过拟合
5. **LoRA 秩的选择**：r=4~8 通常足够用于指令微调；复杂任务可尝试 r=16~32
6. **推理时合并 LoRA 权重**：`model.merge_and_unload()` 合并权重可消除推理时的额外开销

## 总结

| 方法 | 显存需求 (7B) | 训练速度 | 适用场景 |
|------|-------------|---------|---------|
| 全参数微调 | ~56 GB (FP16) | 慢 | 机构级训练 |
