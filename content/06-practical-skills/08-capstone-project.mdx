# 第三步：Baseline 模型（简单 CNN）

在往复杂方向走之前，先建立一个简单的 baseline。这个 baseline 有两个作用：(1) 确保训练流程正确；(2) 提供性能下界。

```python
class BaselineCNN(nn.Module):
    """简单的 CNN Baseline——如果这都训不好，说明数据或流程有问题"""
    def __init__(self, num_classes=10):
        super().__init__()
        self.conv1 = nn.Conv2d(3, 32, 3, padding=1)
        self.conv2 = nn.Conv2d(32, 64, 3, padding=1)
        self.pool = nn.MaxPool2d(2, 2)
        self.fc1 = nn.Linear(64 * 8 * 8, 256)
        self.fc2 = nn.Linear(256, num_classes)
        self.dropout = nn.Dropout(0.3)

    def forward(self, x):
        x = self.pool(F.relu(self.conv1(x)))
        x = self.pool(F.relu(self.conv2(x)))
        x = x.view(x.size(0), -1)
        x = F.relu(self.fc1(x))
        x = self.dropout(x)
        x = self.fc2(x)
        return x


def train_epoch(model, loader, criterion, optimizer, device):
    model.train()
    running_loss, correct, total = 0.0, 0, 0
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        running_loss += loss.item()
        _, preds = outputs.max(1)
        correct += preds.eq(labels).sum().item()
        total += labels.size(0)
    return running_loss / len(loader), correct / total


def validate(model, loader, criterion, device):
    model.eval()
    running_loss, correct, total = 0.0, 0, 0
    all_preds, all_labels = [], []
    with torch.no_grad():
        for images, labels in loader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            loss = criterion(outputs, labels)
            running_loss += loss.item()
            _, preds = outputs.max(1)
            correct += preds.eq(labels).sum().item()
            total += labels.size(0)
            all_preds.extend(preds.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
    return running_loss / len(loader), correct / total, all_preds, all_labels


def train_model(model, train_loader, val_loader, epochs=30, lr=0.001,
                patience=5, device='cpu'):
    """完整的训练函数，带早停"""
    model = model.to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=lr)
    scheduler = optim.lr_scheduler.ReduceLROnPlateau(
        optimizer, mode='min', factor=0.5, patience=3, verbose=True
    )

    history = {'train_loss': [], 'train_acc': [],
               'val_loss': [], 'val_acc': []}
    best_val_acc = 0
    best_model_state = None
    patience_counter = 0

    for epoch in range(epochs):
        train_loss, train_acc = train_epoch(
            model, train_loader, criterion, optimizer, device
        )
        val_loss, val_acc, _, _ = validate(
            model, val_loader, criterion, device
        )
        scheduler.step(val_loss)

        history['train_loss'].append(train_loss)
        history['train_acc'].append(train_acc)
        history['val_loss'].append(val_loss)
        history['val_acc'].append(val_acc)

        print(f"Epoch {epoch+1:2d}/{epochs} | "
              f"Train Loss: {train_loss:.4f} | Train Acc: {train_acc:.4f} | "
              f"Val Loss: {val_loss:.4f} | Val Acc: {val_acc:.4f}")

        if val_acc > best_val_acc:
            best_val_acc = val_acc
            best_model_state = model.state_dict().copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                print(f"早停! 最佳验证准确率: {best_val_acc:.4f}")
                break

    model.load_state_dict(best_model_state)
    return model, history
```

# 第四步：改进模型

基于 baseline，我们设计一个更深的 CNN，加入 Batch Normalization 和残差连接式设计。

```python
class ImprovedCNN(nn.Module):
    """改进的 CNN：更深 + BatchNorm + 更好的架构"""
    def __init__(self, num_classes=10):
        super().__init__()

        # Block 1
        self.conv_block1 = nn.Sequential(
            nn.Conv2d(3, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.1)
        )

        # Block 2
        self.conv_block2 = nn.Sequential(
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.Conv2d(128, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.2)
        )

        # Block 3
        self.conv_block3 = nn.Sequential(
            nn.Conv2d(128, 256, 3, padding=1), nn.BatchNorm2d(256), nn.ReLU(),
            nn.Conv2d(256, 256, 3, padding=1), nn.BatchNorm2d(256), nn.ReLU(),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.3)
        )

        # 分类头
        self.classifier = nn.Sequential(
            nn.AdaptiveAvgPool2d((1, 1)),
            nn.Flatten(),
            nn.Linear(256, 128), nn.ReLU(), nn.Dropout(0.4),
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.conv_block1(x)
        x = self.conv_block2(x)
        x = self.conv_block3(x)
        x = self.classifier(x)
        return x
```

# 第六步：错误分析

找出模型在哪里犯错，这个信息比总体准确率更有价值。

```python
def error_analysis(model, dataloader, device, class_names, top_n=10):
    """分析模型的错误预测"""
    model.eval()
    errors = []  # (image, true_label, pred_label, confidence)

    with torch.no_grad():
        for images, labels in dataloader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            probs = F.softmax(outputs, dim=1)
            preds = outputs.argmax(1)

            mask = preds != labels
            for i in range(len(mask)):
                if mask[i]:
                    errors.append({
                        'image': images[i].cpu(),
                        'true': labels[i].item(),
                        'predicted': preds[i].item(),
                        'confidence': probs[i, preds[i]].item()
                    })

    # 分析混淆最多的类别对
    confusion_pairs = defaultdict(int)
    for e in errors:
        pair = (class_names[e['true']], class_names[e['predicted']])
        confusion_pairs[pair] += 1

    print("最容易混淆的类别对 (TOP 10):")
    for (true_cls, pred_cls), count in sorted(
        confusion_pairs.items(), key=lambda x: x[1], reverse=True
    )[:top_n]:
        print(f"  {true_cls:>10s} → {pred_cls:<10s}: {count} 次")

    # 各类别准确率
    per_class_correct = defaultdict(int)
    per_class_total = defaultdict(int)
    for e in errors:
        per_class_total[e['true']] += 1
    # 重新计算（包括正确的）
    model.eval()
    with torch.no_grad():
        for images, labels in dataloader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            preds = outputs.argmax(1)
            for i in range(len(labels)):
                per_class_total[labels[i].item()] += 1
                if preds[i] == labels[i]:
                    per_class_correct[labels[i].item()] += 1

    print("\n各类别准确率:")
    for i, name in enumerate(class_names):
        total = per_class_total[i]
        correct = per_class_correct[i]
        acc = correct / total if total > 0 else 0
        bar = '█' * int(acc * 20) + '░' * (20 - int(acc * 20))
        print(f"  {name:>10s}: {bar} {acc:.4f} ({correct}/{total})")

    # 找出高置信度错误
    errors.sort(key=lambda x: x['confidence'], reverse=True)
    print(f"\n模型最自信的错误 (TOP 5):")
    for i, e in enumerate(errors[:5]):
        print(f"  #{i+1}: 真实={class_names[e['true']]}, "
              f"预测={class_names[e['predicted']]}, 置信度={e['confidence']:.4f}")

    return errors
```

```python
# ---- FastAPI 应用 ----
app = FastAPI(
    title="CIFAR-10 图像分类 API",
    description="本项目综合实战训练的模型推理服务",
    version="1.0.0"
)

CLASSES = ['airplane', 'automobile', 'bird', 'cat', 'deer',
           'dog', 'frog', 'horse', 'ship', 'truck']

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model = None

# 图像预处理
transform = transforms.Compose([
    transforms.Resize((32, 32)),
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.4914, 0.4822, 0.4465],
                         std=[0.2470, 0.2435, 0.2616])
])


class PredictionResult(BaseModel):
    predicted_class: str
    class_index: int
    confidence: float
    top_3_predictions: List[dict]


@app.on_event("startup")
async def load_model():
    global model
    model = ImprovedCNN(num_classes=10)
    checkpoint = torch.load('cifar10_model_final.pt',
                            map_location=device, weights_only=True)
    model.load_state_dict(checkpoint['model_state_dict'])
    model.to(device)
    model.eval()
    print(f"模型已加载 (验证准确率: {checkpoint['val_accuracy']:.4f}, "
          f"测试准确率: {checkpoint['test_accuracy']:.4f})")


@app.get("/health")
async def health():
    return {"status": "healthy", "device": str(device)}


@app.post("/predict", response_model=PredictionResult)
async def predict(file: UploadFile = File(...)):
    if model is None:
        raise HTTPException(status_code=503, detail="模型未加载")

    # 读取并预处理图片
    contents = await file.read()
    image = Image.open(io.BytesIO(contents)).convert('RGB')
    input_tensor = transform(image).unsqueeze(0).to(device)

    # 推理
    with torch.no_grad():
        logits = model(input_tensor)
        probs = F.softmax(logits, dim=1)

    # 获取 top-3 预测
    top3_probs, top3_indices = torch.topk(probs, 3, dim=1)

    return PredictionResult(
        predicted_class=CLASSES[top3_indices[0, 0].item()],
        class_index=top3_indices[0, 0].item(),
        confidence=round(top3_probs[0, 0].item(), 4),
        top_3_predictions=[
            {
                "class": CLASSES[idx.item()],
                "confidence": round(prob.item(), 4)
            }
            for idx, prob in zip(top3_indices[0], top3_probs[0])
        ]
    )
```

# 关键要点

1. **Baseline 先行**：在引入任何复杂技巧前，先用简单模型验证整个流程。
2. **数据增强优先于模型改进**：对 CIFAR-10 这种小数据集，数据增强带来的收益远超加深网络。
3. **BatchNorm + 合理的初始化**几乎消除了训练深度 CNN 的大部分困难。
4. **学习率调度**（如 ReduceLROnPlateau）能让训练最后几个 epoch 的精度再提升 1-2%。
5. **错误分析**：了解模型在哪里犯错（哪些类别容易混淆）比单纯追求高准确率更有实际意义。
6. **部署思维从训练开始**：训练时就记录预处理参数、类别名称等元信息，部署时才不会手忙脚乱。
