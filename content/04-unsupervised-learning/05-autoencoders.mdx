# 自编码器与变分自编码器

自编码器（Autoencoder）是一种用于学习数据高效表示的无监督神经网络。它可以被视为用神经网络实现的降维方法，但比 PCA 更强大——它能学习非线性的压缩映射。

## 自编码器架构

自编码器由三部分组成：

1. **编码器（Encoder）**：将输入 $x \in \mathbb{R}^d$ 映射到潜在表示 $z \in \mathbb{R}^k$（通常 $k \ll d$）
2. **瓶颈（Bottleneck）/潜在空间（Latent Space）**：压缩后的低维表示 $z$
3. **解码器（Decoder）**：从潜在表示 $z$ 重建输入 $\hat{x} \in \mathbb{R}^d$

训练目标是最小化重构误差：

$$\mathcal{L} = \frac{1}{n} \sum_{i=1}^{n} \|x_i - \hat{x}_i\|^2 = \frac{1}{n} \sum_{i=1}^{n} \|x_i - D(E(x_i))\|^2$$

对于二值数据（如 MNIST 将像素归一化到 $[0,1]$），也可使用二值交叉熵（BCE）作为损失函数。

## PyTorch 实现：基础自编码器

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader
from torchvision import datasets, transforms
import matplotlib.pyplot as plt

class Autoencoder(nn.Module):
    """基础全连接自编码器"""
    def __init__(self, input_dim=784, latent_dim=32):
        super().__init__()
        # 编码器
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 128),
            nn.ReLU(),
            nn.Linear(128, latent_dim),
        )
        # 解码器
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 256),
            nn.ReLU(),
            nn.Linear(256, input_dim),
            nn.Sigmoid(),  # 将输出限制在 [0, 1]，匹配归一化像素值
        )

    def forward(self, x):
        z = self.encoder(x)
        x_hat = self.decoder(z)
        return x_hat, z

    def encode(self, x):
        return self.encoder(x)

# 加载 MNIST
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Lambda(lambda x: x.view(-1)),  # 展平为 784 维
])

train_dataset = datasets.MNIST(
    root='./data', train=True, transform=transform, download=True
)
train_loader = DataLoader(train_dataset, batch_size=128, shuffle=True)
```

### 训练自编码器

自编码器的特殊性在于：**输入即标签**。网络不使用人工标注，而是将原始输入作为监督信号：

```python
def train_autoencoder(model, train_loader, epochs=20, lr=1e-3,
                      device='cuda' if torch.cuda.is_available() else 'cpu'):
    """训练自编码器"""
    model = model.to(device)
    optimizer = optim.Adam(model.parameters(), lr=lr)
    criterion = nn.MSELoss()
    losses = []

    for epoch in range(epochs):
        epoch_loss = 0.0
        for batch_idx, (data, _) in enumerate(train_loader):
            data = data.to(device)  # shape: (batch, 784)

            # 前向传播
            reconstructed, latent = model(data)
            loss = criterion(reconstructed, data)  # 输入 = 目标！

            # 反向传播
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            epoch_loss += loss.item()

        avg_loss = epoch_loss / len(train_loader)
        losses.append(avg_loss)
        print(f"Epoch [{epoch+1}/{epochs}], "
              f"Loss: {avg_loss:.4f}")

    return losses
```

## 去噪自编码器（Denoising Autoencoder）

去噪自编码器故意向输入添加噪声，但要求网络重构出**干净**的原始图像。这使得编码器学到的表示更加鲁棒：

```python
class DenoisingAutoencoder(nn.Module):
    """去噪自编码器：输入加噪，输出干净"""
    def __init__(self, input_dim=784, latent_dim=64, noise_factor=0.3):
        super().__init__()
        self.noise_factor = noise_factor
        # 与基础自编码器相同的架构
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 128),
            nn.ReLU(),
            nn.Linear(128, latent_dim),
        )
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 256),
            nn.ReLU(),
            nn.Linear(256, input_dim),
            nn.Sigmoid(),
        )

    def forward(self, x):
        # 向输入添加高斯噪声（仅在训练时）
        if self.training:
            noise = torch.randn_like(x) * self.noise_factor
            x_noisy = x + noise
            x_noisy = torch.clamp(x_noisy, 0., 1.)
        else:
            x_noisy = x

        z = self.encoder(x_noisy)
        x_hat = self.decoder(z)
        return x_hat, z, x_noisy

# 展示去噪效果
def visualize_denoising(model, test_loader, device, n=8):
    """展示去噪自编码器的输入（含噪）、输出（去噪后）与原始图像"""

```

## 变分自编码器（VAE）

标准自编码器将输入映射到潜在空间的一个确定点。VAE 则学习一个概率分布 $q_\phi(z|x)$，将输入映射到潜在空间的一个**区域**。这使得 VAE 成为一个生成模型——我们可以从 $\mathcal{N}(0, I)$ 中采样 $z$，然后通过解码器生成新样本。

### VAE 的核心公式

编码器输出潜在分布的均值和方差：$(\mu, \log\sigma^2) = E_\phi(x)$

然后通过**重参数化技巧**从中采样：$z = \mu + \sigma \odot \epsilon$，其中 $\epsilon \sim \mathcal{N}(0, I)$

VAE 的损失函数由两部分组成：

$$\mathcal{L}_{\text{VAE}} = \underbrace{\mathbb{E}_{z \sim q_\phi(z|x)}[-\log p_\theta(x|z)]}_{\text{重构损失}} + \underbrace{D_{\text{KL}}(q_\phi(z|x) \| p(z))}_{\text{KL 正则化}}$$

其中 $p(z) = \mathcal{N}(0, I)$ 是先验分布。KL 散度的解析形式为：

$$D_{\text{KL}} = -\frac{1}{2} \sum_{j=1}^{k} \left(1 + \log\sigma_j^2 - \mu_j^2 - \sigma_j^2\right)$$

### VAE 的 PyTorch 实现

```python
class VAE(nn.Module):
    """变分自编码器"""
    def __init__(self, input_dim=784, latent_dim=20):
        super().__init__()
        self.latent_dim = latent_dim

        # 编码器：输出均值和对数方差
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, 512),
            nn.ReLU(),
            nn.Linear(512, 256),
            nn.ReLU(),
        )
        self.fc_mu = nn.Linear(256, latent_dim)       # 均值
        self.fc_logvar = nn.Linear(256, latent_dim)   # 对数方差

        # 解码器
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 512),
            nn.ReLU(),
            nn.Linear(512, input_dim),
            nn.Sigmoid(),
        )

    def encode(self, x):
        h = self.encoder(x)
        mu = self.fc_mu(h)
        logvar = self.fc_logvar(h)
        return mu, logvar

    def reparameterize(self, mu, logvar):
        """重参数化技巧：z = mu + sigma * epsilon"""
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + std * eps

    def forward(self, x):
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        x_hat = self.decoder(z)
        return x_hat, mu, logvar, z

def vae_loss(x, x_hat, mu, logvar):
    """VAE 损失 = 重构损失 + KL 散度"""
    # 重构损失 (BCE 适合 MNIST)
    recon_loss = nn.functional.binary_cross_entropy(
        x_hat, x, reduction='sum'
    )

    # KL 散度
    kl_div = -0.5 * torch.sum(
        1 + logvar - mu.pow(2) - logvar.exp()
    )

    # 总损失 (取平均)
    batch_size = x.size(0)
    return (recon_loss + kl_div) / batch_size

def train_vae(model, train_loader, epochs=30, lr=1e-3, device='cpu'):
    """训练 VAE"""
    model = model.to(device)
    optimizer = optim.Adam(model.parameters(), lr=lr)
    losses = []

    for epoch in range(epochs):
        model.train()
        epoch_loss = 0.0
        for data, _ in train_loader:
            data = data.to(device)

            optimizer.zero_grad()
            x_hat, mu, logvar, z = model(data)
            loss = vae_loss(data, x_hat, mu, logvar)
            loss.backward()
            optimizer.step()

            epoch_loss += loss.item()

        avg_loss = epoch_loss / len(train_loader)
        losses.append(avg_loss)
        print(f"Epoch [{epoch+1}/{epochs}], Loss: {avg_loss:.2f}")

    return losses
```

### 从 VAE 生成新样本

```python
def generate_samples(model, n_samples=16, latent_dim=20, device='cpu'):
    """从标准正态分布采样，生成新图像"""
    model.eval()
    with torch.no_grad():
        z = torch.randn(n_samples, latent_dim).to(device)
        generated = model.decoder(z)
        generated = generated.cpu().view(n_samples, 28, 28)

    # 可视化
    grid_size = int(n_samples ** 0.5)
    fig, axes = plt.subplots(grid_size, grid_size,
                             figsize=(2 * grid_size, 2 * grid_size))
    for i, ax in enumerate(axes.flat):
        ax.imshow(generated[i], cmap='gray')
        ax.axis('off')
    plt.suptitle('VAE 生成样本')
    plt.tight_layout()
    plt.show()
    return generated
```

## 应用场景

- **异常检测**：对正常数据训练自编码器，测试时重构误差大的样本即为异常
- **图像去噪**：去噪自编码器直接从含噪图像恢复清晰图像
- **特征学习**：自编码器的潜在表示可作为下游任务的输入特征（预训练）
- **数据生成**：VAE 是生成模型，可生成与训练数据分布相似的新样本
- **数据压缩**：使用比 PCA 更小的瓶颈实现高质量的有损压缩

## 小结

自编码器将降维的自然想法带入了深度学习世界。基础自编码器教会网络"压缩-重建"的映射，去噪自编码器强化了鲁棒性，而 VAE 更将整个潜在空间变成了一个生成引擎。在实践中，对于纯粹的降维需求，先从 PCA 开始；当需要非线性映射或生成能力时，再转向自编码器家族。下一课我们将学习 GANs——当前最强大的生成模型之一。
