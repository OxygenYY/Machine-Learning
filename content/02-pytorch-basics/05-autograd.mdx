## 什么是自动微分？

在深度学习中，我们需要计算损失函数对模型参数的梯度，以便使用梯度下降法更新参数。有三种计算导数的方法：

### 三种微分方式对比

```python
import torch
import time

# 目标：计算 f(x) = x^3 + 2x^2 - 3x + 1 在 x=2 处的导数
# 真实导数：f'(x) = 3x^2 + 4x - 3, f'(2) = 17

def f(x):
    return x**3 + 2*x**2 - 3*x + 1

# === 方法1：手动推导（符号微分）===
# 人工计算导数公式，直接代入
def f_prime_manual(x):
    return 3*x**2 + 4*x - 3

print("=== 三种微分方式计算 f'(2) ===\n")
print(f"1. 手动推导: f'(2) = {f_prime_manual(torch.tensor(2.0))}")

# === 方法2：数值微分（有限差分法）===
# f'(x) ≈ (f(x+h) - f(x)) / h
def numerical_derivative(f, x, h=1e-6):
    return (f(x + h) - f(x)) / h

x_val = torch.tensor(2.0)
num_deriv = numerical_derivative(f, x_val)
print(f"2. 数值微分: f'(2) ≈ {num_deriv:.6f} (近似值，有舍入误差)")

# === 方法3：自动微分（Autograd）===
x = torch.tensor(2.0, requires_grad=True)
y = f(x)
y.backward()  # 自动计算梯度
print(f"3. 自动微分: f'(2) = {x.grad} (精确值，自动计算)")
```

### 三种方法的特点

| 特性 | 手动/符号微分 | 数值微分 | 自动微分 (Autograd) |
|------|-------------|---------|-------------------|
| 精确度 | 精确 | 近似（有截断/舍入误差） | 精确（机器精度） |
| 计算复杂度 | O(1) | O(n)（每个参数需要两次前向） | O(1) |
| 实现难度 | 手动推导，易错 | 简单 | 框架自动处理 |
| 适用场景 | 简单函数 | 验证梯度正确性 | 深度学习（复杂模型） |

## 计算图（Computational Graph）

PyTorch 使用动态计算图来跟踪张量上的操作，从而实现自动微分。

### 计算图的构成

计算图是一个有向无环图（DAG），其中：
- **节点**：张量（数据）和操作（函数）
- **边**：数据的流向

```python
import torch

# 示例：计算 z = (x * y) + (x + y)
# 计算图结构：
#        x ──┬── * ── w ──┐
#        y ──┘              ├── z
#        x ──┬── + ── v ──┘
#        y ──┘

x = torch.tensor(2.0, requires_grad=True)
y = torch.tensor(3.0, requires_grad=True)

w = x * y      # w = 6
v = x + y      # v = 5
z = w + v      # z = 11

print(f"x={x.item()}, y={y.item()}")
print(f"w = x*y = {w.item()}")
print(f"v = x+y = {v.item()}")
print(f"z = w+v = {z.item()}")
print(f"\nz 的 grad_fn: {z.grad_fn}")  # AddBackward0
print(f"w 的 grad_fn: {w.grad_fn}")    # MulBackward0
print(f"v 的 grad_fn: {v.grad_fn}")    # AddBackward0

# 反向传播
z.backward()

# dz/dx = dz/dw * dw/dx + dz/dv * dv/dx = 1 * y + 1 * 1 = y + 1 = 4
print(f"\nx.grad (dz/dx): {x.grad}")  # 应为 4.0
# dz/dy = dz/dw * dw/dy + dz/dv * dv/dy = 1 * x + 1 * 1 = x + 1 = 3
print(f"y.grad (dz/dy): {y.grad}")  # 应为 3.0
```

### 前向传播与反向传播

```python
import torch

# 详细演示 forward / backward 过程
def demonstrate_computational_graph():
    # 定义输入
    a = torch.tensor(1.0, requires_grad=True)
    b = torch.tensor(2.0, requires_grad=True)

    # 前向传播：逐步构建计算图
    print("=== 前向传播（构建计算图）===")
    c = a * b
    print(f"c = a * b = {c.item()}, grad_fn: {c.grad_fn}")

    d = c + a
    print(f"d = c + a = {d.item()}, grad_fn: {d.grad_fn}")

    e = d ** 2
    print(f"e = d^2 = {e.item()}, grad_fn: {e.grad_fn}")

    # 反向传播：沿计算图反向传播梯度
    print("\n=== 反向传播（计算梯度）===")
    e.backward()

    # e = d^2 => de/dd = 2d = 2*3 = 6
    # d = c + a => dd/dc = 1, dd/da = 1
    # c = a * b => dc/da = b = 2, dc/db = a = 1

    # de/da = de/dd * dd/da + de/dd * dd/dc * dc/da
    #       = 6 * 1 + 6 * 1 * 2 = 6 + 12 = 18
    print(f"de/da (a.grad): {a.grad}")  # 18

    # de/db = de/dd * dd/dc * dc/db
    #       = 6 * 1 * 1 = 6
    print(f"de/db (b.grad): {b.grad}")  # 6

demonstrate_computational_graph()
```

## requires_grad：梯度追踪

`requires_grad` 属性决定了 PyTorch 是否需要为该张量记录梯度。

```python
import torch

# === 默认行为 ===
# 用户创建的张量默认 requires_grad=False
x_default = torch.randn(3, 3)
print(f"默认 requires_grad: {x_default.requires_grad}")

# 显式设置 requires_grad=True
x = torch.randn(3, 3, requires_grad=True)
print(f"显式设置后: {x.requires_grad}")

# 也可以在创建后修改（使用 requires_grad_()）
y = torch.randn(3, 3)
y.requires_grad_(True)  # 注意下划线，原地操作
print(f"requires_grad_() 后: {y.requires_grad}")

# === 梯度追踪的传播规则 ===
# 如果任何一个输入 requires_grad=True，输出也会 requires_grad=True
a = torch.randn(2, requires_grad=True)
b = torch.randn(2, requires_grad=False)

c = a + b
print(f"\na.requires_grad={a.requires_grad}, b.requires_grad={b.requires_grad}")
print(f"c=a+b 的 requires_grad: {c.requires_grad}")  # True（一个输入有梯度就传播）

# 只有所有输入都不需要梯度时，输出才不需要梯度
d = b * 2
print(f"d=b*2 的 requires_grad: {d.requires_grad}")  # False

# === 叶张量 vs 非叶张量 ===
w = torch.tensor([1.0, 2.0, 3.0], requires_grad=True)
v = w * 2  # v 不是叶张量

print(f"\nw 是叶张量: {w.is_leaf}")  # True
print(f"v 是叶张量: {v.is_leaf}")   # False
print(f"w 有 .grad: {w.grad is None}")  # w.grad 目前为 None（尚未调用 backward）
```

## backward() 和 .grad

```python
import torch

# === 基础示例：f(x) = x^2 在 x=3 处的梯度 ===
x = torch.tensor(3.0, requires_grad=True)
f = x ** 2         # f(x) = x^2
f.backward()       # 计算 df/dx
print(f"f(x)=x^2, x=3")
print(f"f(3) = {f.item()}")
print(f"f'(3) = 2*x = {x.grad.item()}")  # 6.0

# === 多变量函数 ===
x1 = torch.tensor(2.0, requires_grad=True)
x2 = torch.tensor(3.0, requires_grad=True)

y = x1**3 + 2 * x1 * x2 + x2**2  # y = x1^3 + 2*x1*x2 + x2^2
# dy/dx1 = 3*x1^2 + 2*x2 = 3*4 + 6 = 18
# dy/dx2 = 2*x1 + 2*x2 = 4 + 6 = 10

y.backward()
print(f"\n多变量函数:")
print(f"y = x1^3 + 2*x1*x2 + x2^2")
print(f"dy/dx1 = {x1.grad}")  # 18
print(f"dy/dx2 = {x2.grad}")  # 10

# === backward() 只能对标量调用 ===
# 如果 y 是向量/矩阵，需要传入 gradient 参数
x = torch.randn(3, requires_grad=True)
y = x * 2  # y 是向量，shape (3,)

try:
    y.backward()  # 报错！backward() 只能用于标量
except RuntimeError as e:
    print(f"\n向量调用 backward 报错: {type(e).__name__}")

# 正确做法：将向量转为标量（如求和），或传入 gradient 参数
# 方法1：转为标量
y.sum().backward()
print(f"y.sum().backward() 后 x.grad: {x.grad}")

# 方法2：传入 gradient 参数
x = torch.randn(3, requires_grad=True)
y = x * 2
# 传入与 y 同形状的 gradient（表示"外部梯度"）
y.backward(gradient=torch.ones_like(y))
print(f"y.backward(ones_like) 后 x.grad: {x.grad}")

# backward() 保留图，允许多次反向传播
x = torch.tensor(3.0, requires_grad=True)
y = x ** 2
y.backward(retain_graph=True)  # 保留计算图
print(f"\n第一次 backward: x.grad = {x.grad}")  # 6
y.backward(retain_graph=True)
print(f"第二次 backward: x.grad = {x.grad}")  # 12（梯度累积！）
```

## torch.no_grad()：禁用梯度追踪

在推理和评估阶段，不需要计算梯度。使用 `torch.no_grad()` 可以显著节省内存和计算。

```python
import torch

# === 为什么需要 no_grad？===
# 1. 节省内存 - 不需要存储中间激活值（计算图）
# 2. 加速计算 - 跳过梯度计算
# 3. 避免意外修改参数梯度

# 创建一个需要梯度的张量
x = torch.tensor(2.0, requires_grad=True)

# 在 no_grad 上下文中的操作不会被追踪
with torch.no_grad():
    y = x * 3 + 5  # 这些操作不会构建计算图
    print(f"在 no_grad 中 y.requires_grad: {y.requires_grad}")
    print(f"在 no_grad 中 y.grad_fn: {y.grad_fn}")

# 上下文外的操作正常追踪
z = x * 3 + 5
print(f"正常模式 z.requires_grad: {z.requires_grad}")
print(f"正常模式 z.grad_fn: {z.grad_fn}")

# === 实际应用场景 ===
print("\n=== 推理/评估场景 ===")
# 模拟一个简单的"模型"
def model(x):
    return x ** 2 + 3 * x + 1

# 训练时：需要梯度
x_train = torch.tensor([1.0, 2.0, 3.0], requires_grad=True)
pred_train = model(x_train)
loss = pred_train.sum()
loss.backward()
print(f"训练模式: x.grad = {x_train.grad}")

# 推理时：不需要梯度（更快，更省内存）
x_eval = torch.tensor([1.0, 2.0, 3.0])
with torch.no_grad():
    pred_eval = model(x_eval)
    print(f"推理模式: pred = {pred_eval}")

# === torch.set_grad_enabled() ===
# 可以用条件控制梯度开关
is_training = False
with torch.set_grad_enabled(is_training):
    pred = model(torch.tensor([1.0, 2.0, 3.0]))
    print(f"set_grad_enabled({is_training}): requires_grad={pred.requires_grad}")
```

## detach()：从计算图中分离

`detach()` 创建一个与原张量共享数据但不参与梯度计算的新张量。

```python
import torch

# === detach() 基本用法 ===
x = torch.tensor(2.0, requires_grad=True)
y = x ** 2          # y 在计算图中
y_detached = y.detach()  # 共享数据，但不在计算图中

print(f"y (原始):      value={y.item()}, requires_grad={y.requires_grad}")
print(f"y_detached:    value={y_detached.item()}, requires_grad={y_detached.requires_grad}")

# 修改 detach 后的数据不会影响原张量的梯度计算
z = y_detached * 3   # 这个操作不会连接到 x
# z.backward()       # 报错：z 不在计算图中

# 正确：通过 y 计算梯度
y.backward()
print(f"\ny.backward() 后 x.grad: {x.grad}")  # 4 (dy/dx = 2x = 4)

# === detach() 的应用场景 ===

# 场景1：固定预训练模型的参数
print("\n=== 场景1：冻结部分参数 ===")
# 模拟：feature_extractor 的输出需要 detach，不传梯度
features = torch.randn(5, 10, requires_grad=True)
# 假设 feature_extractor 是预训练的，我们不想更新它
detached_features = features.detach()
# 后续的分类器可以根据 detached_features 训练
classifier_output = detached_features.sum(dim=1)
# 梯度不会回流到 features

# 场景2：计算指标时不影响梯度
print("\n=== 场景2：记录指标 ===")
w = torch.tensor([1.0, 2.0], requires_grad=True)
loss = (w ** 2).sum()
# 我们想记录准确率，但不希望它影响梯度
accuracy = (w > 1.5).float().mean().detach()  # 不影响梯度
loss.backward()  # accuracy 不影响这个梯度计算
print(f"w.grad = {w.grad}")

# 场景3：GAN 中训练生成器时固定判别器
print("\n=== 场景3：GAN 训练模式 ===")
# 简化 GAN 示例
def discriminator(x):
    # 简化的判别器
    return x.sum()

gen_output = torch.randn(3, requires_grad=True)  # 生成器的输出

# 训练判别器时：不追踪生成器的梯度
d_real = discriminator(torch.randn(3))  # 真实样本
d_fake = discriminator(gen_output.detach())  # 生成样本，但梯度不流向生成器
d_loss = (d_real - d_fake) ** 2
# d_loss.backward()  # 只更新判别器，不影响生成器

# 训练生成器时：梯度需要流过判别器
g_loss = -discriminator(gen_output)  # 梯度流过生成器
# g_loss.backward()  # 更新生成器
```

## 梯度累积与 zero_grad()

这是初学者最常遇到的"坑"之一：PyTorch 默认会累加梯度，而不是覆盖。

```python
import torch

# 创建一个简单参数
w = torch.tensor(2.0, requires_grad=True)

print("=== 梯度累积演示 ===\n")

# 第1次计算
loss1 = w ** 2       # f(w) = w^2
loss1.backward()     # df/dw = 2w = 4
print(f"第1次: loss1={loss1.item()}, w.grad={w.grad}")

# 第2次计算（不归零梯度）
loss2 = (w - 1) ** 2 # f(w) = (w-1)^2
loss2.backward()     # df/dw = 2(w-1) = 2，但累加到之前的 4 上
print(f"第2次: loss2={loss2.item()}, w.grad={w.grad} (4 + 2 = 6)")

# 第3次计算（不归零梯度）
loss3 = (w - 3) ** 2 # f(w) = (w-3)^2
loss3.backward()     # df/dw = 2(w-3) = -2，累加
print(f"第3次: loss3={loss3.item()}, w.grad={w.grad} (6 + (-2) = 4)")

# === 正确的训练循环 ===
print("\n=== 正确的训练循环（每次迭代归零梯度）===")

w = torch.tensor(2.0, requires_grad=True)

for step in range(3):
    # 1. 归零梯度（这一步经常被遗忘！）
    if w.grad is not None:
        w.grad.zero_()

    # 2. 前向传播
    loss = (w - 1.5) ** 2

    # 3. 反向传播
    loss.backward()

    # 4. 更新参数
    lr = 0.1
    with torch.no_grad():
        w -= lr * w.grad  # 梯度下降

    print(f"Step {step+1}: w={w.item():.4f}, loss={loss.item():.4f}, grad={w.grad.item():.4f}")

# === 梯度累积的实际用途 ===
print("\n=== 梯度累积：模拟大 batch size ===")

w = torch.tensor(2.0, requires_grad=True)
accumulation_steps = 4  # 模拟 batch_size * 4

for micro_step in range(accumulation_steps):
    # 每个 micro-batch 计算损失
    loss = (w - 3.0) ** 2 / accumulation_steps  # 注意除以累积步数
    loss.backward()
    print(f"Micro-step {micro_step+1}: grad={w.grad.item():.4f}")

# 累积完成后一次更新
with torch.no_grad():
    w -= 0.1 * w.grad
print(f"\n累积 {accumulation_steps} 步后:")
print(f"  总梯度: {w.grad.item():.4f}")
print(f"  更新后 w: {w.item():.4f}")

# 如果不除以 accumulation_steps，梯度会偏大
```

## 完整的机器学习示例

让我们用自动微分实现一个简单的线性回归，将理论串联起来。

```python
import torch
import matplotlib.pyplot as plt

# 设置随机种子
torch.manual_seed(42)

# === 1. 生成训练数据 ===
# 真实关系：y = 3x + 2 + 噪声
true_w = 3.0
true_b = 2.0
n_samples = 100

X = torch.randn(n_samples, 1) * 2  # 输入特征，标准正态分布拉伸
noise = torch.randn(n_samples, 1) * 0.5
y = true_w * X + true_b + noise     # 标签

print(f"训练数据: X.shape={X.shape}, y.shape={y.shape}")
print(f"真实参数: w={true_w}, b={true_b}")

# === 2. 初始化模型参数 ===
w = torch.randn(1, requires_grad=True)
b = torch.zeros(1, requires_grad=True)

print(f"\n初始参数: w={w.item():.4f}, b={b.item():.4f}")

# === 3. 定义模型和损失函数 ===
def linear_model(x, w, b):
    return x @ w.T + b  # 或 x * w + b

def mse_loss(y_pred, y_true):
    return ((y_pred - y_true) ** 2).mean()

# === 4. 训练循环 ===
learning_rate = 0.01
n_epochs = 100

loss_history = []
w_history = []
b_history = []

for epoch in range(n_epochs):
    # 前向传播
    y_pred = linear_model(X, w, b)
    loss = mse_loss(y_pred, y)

    # 反向传播
    loss.backward()

    # 更新参数
    with torch.no_grad():
        w -= learning_rate * w.grad
        b -= learning_rate * b.grad

    # 归零梯度
    w.grad.zero_()
    b.grad.zero_()

    # 记录历史
    loss_history.append(loss.item())
    w_history.append(w.item())
    b_history.append(b.item())

    if (epoch + 1) % 20 == 0:
        print(f"Epoch {epoch+1:3d}: loss={loss.item():.6f}, "
              f"w={w.item():.4f}, b={b.item():.4f}")

# === 5. 最终结果 ===
print(f"\n=== 训练完成 ===")
print(f"学习到的参数: w={w.item():.4f} (真实值: {true_w})")
print(f"              b={b.item():.4f} (真实值: {true_b})")
print(f"最终损失:     {loss_history[-1]:.6f}")
print(f"初始损失:     {loss_history[0]:.6f}")

# === 6. 梯度流分析 ===
# 可以手动验证梯度
with torch.no_grad():
    y_pred = linear_model(X, w, b)
    # d(Loss)/dw = 2/N * sum((y_pred - y) * x)
    manual_dw = (2 / n_samples) * ((y_pred - y) * X).sum()
    # d(Loss)/db = 2/N * sum((y_pred - y))
    manual_db = (2 / n_samples) * (y_pred - y).sum()

print(f"\n手动计算的梯度: dw={manual_dw.item():.4f}, db={manual_db.item():.4f}")
```

## 本节小结

- 自动微分比手动推导更自动、比数值微分更精确高效
- PyTorch 通过**动态计算图**记录操作，`backward()` 时自动计算梯度
- `requires_grad=True` 开启梯度追踪，输出张量的 `grad_fn` 记录了创建它的操作
- `backward()` 计算梯度并存储在 `.grad` 属性中，默认会**累加**梯度
- `torch.no_grad()` 禁用梯度追踪，用于推理和评估
- `detach()` 创建共享数据但脱离计算图的新张量
- 训练循环中每次迭代必须调用 `.zero_()` 归零梯度，否则梯度会累积
- 梯度累积可以模拟大 batch size 的训练
