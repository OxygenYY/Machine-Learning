## 索引和切片

PyTorch 张量的索引和切片语法与 NumPy 基本一致。如果你熟悉 NumPy 的索引方式，可以直接应用到 PyTorch 中。

```python
import torch

# 创建一个 3x4 的张量
x = torch.arange(12).reshape(3, 4)
print(f"原始张量 x (3x4):\n{x}\n")

# === 基础索引 ===
print(f"x[0]:    第一行 = {x[0]}")         # 第0行
print(f"x[1, 2]: 第1行第2列 = {x[1, 2]}") # 单个元素
print(f"x[-1]:   最后一行 = {x[-1]}")       # 负数索引
print(f"x[1:]:   第1行及之后:\n{x[1:]}\n") # 切片

# === 多维切片 ===
print(f"x[:, 1:3]: 所有行，第1-2列:\n{x[:, 1:3]}\n")
print(f"x[0:2, ::2]: 前两行，每隔一列:\n{x[0:2, ::2]}\n")

# === 高级索引 ===
# 整数数组索引
rows = torch.tensor([0, 2])
cols = torch.tensor([1, 3])
print(f"x[[0,2], [1,3]]: {x[rows, cols]}")  # 取 (0,1) 和 (2,3)

# 布尔索引（掩码索引）
mask = x > 5
print(f"\n大于5的元素:\n{mask}")
print(f"大于5的元素值: {x[mask]}")
```

### 高级索引技巧

```python
import torch

x = torch.arange(24).reshape(2, 3, 4)  # 3D 张量
print(f"3D 张量 x (2x3x4):\n{x}\n")

# 用 ... 表示"所有剩下的维度"
print(f"x[0, ...]: 取第0个批次的所有数据:\n{x[0, ...]}\n")
print(f"x[..., 2]: 取所有批次、所有行的第2列:\n{x[..., 2]}\n")

# 混合使用标量和切片
print(f"x[0, 1:, :2]:\n{x[0, 1:, :2]}\n")

# gather - 按索引收集元素（常用于分类任务中取预测值）
values = torch.tensor([[10, 20, 30],
                        [40, 50, 60]])
indices = torch.tensor([[2, 0],     # 取每行指定位置的元素
                         [1, 2]])
gathered = torch.gather(values, dim=1, index=indices)
print(f"gather 结果:\n{gathered}")
# 第0行: 取索引[2, 0] -> [30, 10]
# 第1行: 取索引[1, 2] -> [50, 60]
```

## 形状变换（Reshaping）

在深度学习中，经常需要改变张量的形状以适应不同层的输入要求。PyTorch 提供了多种形状变换操作，各有特点和适用场景。

### view() vs reshape()

```python
import torch

x = torch.arange(12)
print(f"原始张量 x: {x}, shape={x.shape}\n")

# reshape - 总是安全可用，可能返回副本
r1 = x.reshape(3, 4)
print(f"reshape(3, 4):\n{r1}\n")

r2 = x.reshape(2, 2, 3)
print(f"reshape(2, 2, 3):\n{r2}\n")

# -1 表示让 PyTorch 自动推断该维度大小
r3 = x.reshape(3, -1)    # -1 会被推断为 4
print(f"reshape(3, -1): shape={r3.shape}\n")

r4 = x.reshape(-1, 2)    # -1 会被推断为 6
print(f"reshape(-1, 2): shape={r4.shape}\n")

# view - 要求张量在内存中是连续的
x_contiguous = torch.randn(2, 3, 4)
print(f"原张量是否连续: {x_contiguous.is_contiguous()}")

v1 = x_contiguous.view(2, 12)
print(f"view(2, 12): shape={v1.shape}")

# 如果张量不连续，view 会报错
x_transposed = x_contiguous.transpose(0, 1)
print(f"转置后是否连续: {x_transposed.is_contiguous()}")

# 解决方法：先调用 .contiguous()
try:
    x_transposed.view(-1)
except RuntimeError as e:
    print(f"view 报错: {type(e).__name__}")
    # 正确做法
    v2 = x_transposed.contiguous().view(-1)
    print(f"contiguous().view(-1): shape={v2.shape}")

# 经验法则：不确定时用 reshape，性能敏感且确保连续时用 view
```

### 转置和维度重排

```python
import torch

x = torch.arange(6).reshape(2, 3)
print(f"原始张量 (2x3):\n{x}\n")

# transpose - 交换两个维度
t = x.transpose(0, 1)  # 变成 3x2
print(f"transpose(0,1) (3x2):\n{t}\n")

# .T 属性 - 对 2D 张量的转置快捷方式
print(f"x.T (等价于 transpose(0,1)):\n{x.T}\n")

# permute - 任意重排多个维度
x3d = torch.arange(24).reshape(2, 3, 4)
print(f"3D 张量 shape: {x3d.shape}")

p = x3d.permute(2, 0, 1)  # 维度顺序变为 (4, 2, 3)
print(f"permute(2, 0, 1) shape: {p.shape}")

p2 = x3d.permute(1, 2, 0)  # 维度顺序变为 (3, 4, 2)
print(f"permute(1, 2, 0) shape: {p2.shape}\n")

# flatten - 展平为一维
f = x3d.flatten()
print(f"flatten: shape={f.shape}")

# flatten 指定起止维度
f2 = x3d.flatten(start_dim=1)  # 从第1维开始展平
print(f"flatten(start_dim=1): shape={f2.shape}")  # (2, 12)
```

## 拼接和拆分

### 拼接：cat vs stack

```python
import torch

a = torch.tensor([[1, 2], [3, 4]])
b = torch.tensor([[5, 6], [7, 8]])

print(f"a:\n{a}\n")
print(f"b:\n{b}\n")

# cat - 沿已有维度拼接（不增加维度）
c0 = torch.cat([a, b], dim=0)  # 沿行方向拼接
print(f"cat(dim=0) 沿行拼接 (4x2):\n{c0}\n")

c1 = torch.cat([a, b], dim=1)  # 沿列方向拼接
print(f"cat(dim=1) 沿列拼接 (2x4):\n{c1}\n")

# stack - 沿新维度堆叠（会增加一个维度）
s0 = torch.stack([a, b], dim=0)  # 在新第0维堆叠
print(f"stack(dim=0) shape: {s0.shape}")  # (2, 2, 2)
print(f"{s0}\n")

s1 = torch.stack([a, b], dim=1)  # 在新第1维堆叠
print(f"stack(dim=1) shape: {s1.shape}")  # (2, 2, 2)
print(f"{s1}")

# 关键区别：
# cat:   (2,2) + (2,2) -> (4,2) 或 (2,4)，在已有维度上延伸
# stack: (2,2) + (2,2) -> (2,2,2)，创建新维度
```

### 拆分张量

```python
import torch

x = torch.arange(12).reshape(3, 4)
print(f"原始张量 (3x4):\n{x}\n")

# chunk - 按数量等分
chunks = torch.chunk(x, chunks=3, dim=0)  # 沿行分成3份，每份1行
for i, chunk in enumerate(chunks):
    print(f"chunk {i}:\n{chunk}\n")

# split - 按大小拆分（可以不等分）
splits = torch.split(x, split_size_or_sections=[1, 2], dim=0)
# 第0维拆成1行和2行
print(f"split 第1块:\n{splits[0]}")
print(f"split 第2块:\n{splits[1]}")

# 也可均匀拆分
equal_splits = torch.split(x, split_size_or_sections=2, dim=1)
for i, s in enumerate(equal_splits):
    print(f"equal_split {i} (2x2):\n{s}\n")
```

## squeeze 和 unsqueeze

这两个操作用于添加和删除大小为 1 的维度，在处理批次维度时非常常用。

```python
import torch

# 创建一个带有冗余维度的张量
x = torch.arange(6).reshape(1, 6, 1)
print(f"原始张量 shape: {x.shape}")  # (1, 6, 1)

# squeeze - 删除所有大小为1的维度
s = x.squeeze()
print(f"squeeze 后 shape: {s.shape}")  # (6,)

# squeeze(dim) - 删除指定维度（仅当该维度为1时有效）
s0 = x.squeeze(0)   # 删除第0维
print(f"squeeze(0) shape: {s0.shape}")  # (6, 1)

s2 = x.squeeze(2)   # 删除第2维
print(f"squeeze(2) shape: {s2.shape}")  # (1, 6)

# unsqueeze - 在指定位置增加一个大小为1的维度
x_flat = torch.tensor([1, 2, 3])  # shape: (3,)

u0 = x_flat.unsqueeze(0)  # 在第0维增加 -> (1, 3) 变成一行
print(f"\nunsqueeze(0): shape={u0.shape}, 值={u0}")

u1 = x_flat.unsqueeze(1)  # 在第1维增加 -> (3, 1) 变成一列
print(f"unsqueeze(1): shape={u1.shape}, 值=\n{u1}")

# 实际应用：将单个样本扩展为批次
single_image = torch.randn(3, 224, 224)  # 单张图片 (C, H, W)
batch_input = single_image.unsqueeze(0)   # 变成批次 (1, C, H, W)
print(f"\n单张图片 -> 批次输入: {single_image.shape} -> {batch_input.shape}")

# None 索引等价于 unsqueeze
batch_input_alt = single_image[None, ...]  # 等价于 unsqueeze(0)
print(f"用 None 索引: {batch_input_alt.shape}")
```

## 广播机制（Broadcasting）

广播是 PyTorch 对不同形状的张量执行逐元素运算的方式。理解广播规则对于高效编写深度学习代码至关重要。

### 广播规则

两个张量进行运算时，从最后一个维度开始向前比较：

1. 如果两个维度相等，可以运算
2. 如果其中一个维度为 1，广播该维度以匹配另一个
3. 如果其中一个张量维度较少，在其前面补 1

```python
import torch

# === 示例1：标量广播 ===
x = torch.ones(3, 4)
scalar = 2.0
result = x + scalar  # scalar 被广播为 shape (3,4) 的张量
print(f"标量广播: (3,4) + 标量 = shape {result.shape}\n")

# === 示例2：向量广播 ===
a = torch.arange(3).reshape(3, 1)  # shape (3, 1)
b = torch.arange(4).reshape(1, 4)  # shape (1, 4)

# 广播过程：a(3,1) -> a(3,4), b(1,4) -> b(3,4)
c = a + b
print(f"广播加法 (3,1) + (1,4) = (3,4):\n{c}\n")

# === 示例3：广播规则应用 ===
# shape (4, 3) + shape (3,) -> shape (4, 3)
x = torch.ones(4, 3)
y = torch.tensor([1.0, 2.0, 3.0])  # shape (3,)
print(f"(4,3) + (3,) =\n{x + y}\n")

# === 示例4：不兼容的形状 ===
try:
    x = torch.ones(4, 3)
    y = torch.ones(4, 1)
    z = x + y  # (4,3) + (4,1) -> 可以，(4,1) 在第1维广播为3
    print(f"(4,3) + (4,1) 成功: shape={z.shape}")
except RuntimeError as e:
    print(f"错误: {e}")

try:
    x = torch.ones(4, 3)
    y = torch.ones(5,)   # shape (5,)
    z = x + y  # (4,3) + (5,) -> 不兼容！3 != 5
    print(f"结果: {z}")
except RuntimeError as e:
    print(f"(4,3) + (5,) 广播失败: 维度不兼容 (3 != 5)")

# === 广播规则检查工具 ===
# PyTorch 提供 broadcast_shapes 来检查广播后的形状
broadcasted_shape = torch.broadcast_shapes((4, 1, 3), (1, 5, 3), (4, 5, 1))
print(f"\n广播后形状: {broadcasted_shape}")  # (4, 5, 3)
```

### 广播的实际应用

```python
import torch

# 应用1：数据中心化（减去均值）
data = torch.randn(100, 10)        # 100个样本，每个10维
mean = data.mean(dim=0, keepdim=True)  # shape (1, 10)
centered = data - mean             # 广播：每个样本减去均值
print(f"数据中心化: {centered.shape}, 新均值≈0: {centered.mean():.6f}")

# 应用2：图像标准化
images = torch.randn(16, 3, 32, 32)  # 批次归一化
# 假设有预计算的均值和标准差
mean_rgb = torch.tensor([0.485, 0.456, 0.406]).reshape(1, 3, 1, 1)
std_rgb = torch.tensor([0.229, 0.224, 0.225]).reshape(1, 3, 1, 1)
normalized = (images - mean_rgb) / std_rgb
print(f"图像标准化: {normalized.shape}")

# 应用3：注意力机制中的掩码
seq_len = 5
scores = torch.randn(1, seq_len, seq_len)  # 注意力分数
mask = torch.tril(torch.ones(seq_len, seq_len)).bool()  # 下三角掩码
# mask shape (seq_len, seq_len) 被广播为 (1, seq_len, seq_len)
masked_scores = scores.masked_fill(~mask, float('-inf'))
print(f"因果掩码形状: {masked_scores.shape}")
```

## 归约操作（Reduction Operations）

归约操作将张量沿指定维度压缩，减少维度数量。

```python
import torch

x = torch.arange(12, dtype=torch.float32).reshape(3, 4)
print(f"原始张量 x (3x4):\n{x}\n")

# === 基本归约 ===
print(f"sum():      {x.sum():.0f}")        # 所有元素求和
print(f"mean():     {x.mean():.4f}")       # 所有元素均值
print(f"max():      {x.max():.0f}")        # 最大值
print(f"min():      {x.min():.0f}")        # 最小值
print(f"std():      {x.std():.4f}")        # 标准差
print(f"var():      {x.var():.4f}")        # 方差
print(f"prod():     {x.prod():.0f}")       # 所有元素乘积

# === 沿指定维度归约 ===
print(f"\n沿 dim=0 (跨行, 输出形状={x.sum(dim=0).shape}):\n{x.sum(dim=0)}")
print(f"沿 dim=1 (跨列, 输出形状={x.sum(dim=1).shape}):\n{x.sum(dim=1)}")

# keepdim=True 保持维度，便于后续广播
sum_keepdim = x.sum(dim=1, keepdim=True)
print(f"\nsum(dim=1, keepdim=True) shape: {sum_keepdim.shape}")
print(f"结果:\n{sum_keepdim}")

# 用 keepdim 的好处：可以直接用于广播除法
normalized = x / x.sum(dim=1, keepdim=True)  # 每行除以其和
print(f"行归一化结果:\n{normalized}")

# === argmax 和 argmin ===
scores = torch.tensor([[2.1, 5.4, 1.7],
                        [3.3, 0.8, 4.9]])
print(f"\n分数矩阵:\n{scores}")
print(f"每行最大值索引 (argmax dim=1): {scores.argmax(dim=1)}")
print(f"每列最大值索引 (argmax dim=0): {scores.argmax(dim=0)}")

# topk - 取前 k 大/小的值
values, indices = torch.topk(scores, k=2, dim=1)
print(f"\n每行前2大的值:\n{values}")
print(f"前2大的索引:\n{indices}")
```

## 与 NumPy 的互操作

PyTorch 和 NumPy 之间可以无缝转换，但要理解内存共享的机制。

```python
import torch
import numpy as np

# === PyTorch -> NumPy ===
x_torch = torch.tensor([1.0, 2.0, 3.0])

# .numpy() - 返回共享内存的 NumPy 数组
x_np = x_torch.numpy()
print(f"PyTorch 张量: {x_torch}")
print(f"NumPy 数组:   {x_np}")

# 修改 NumPy 会影响原张量（共享内存）
x_np[0] = 99.0
print(f"\n修改 NumPy 后:")
print(f"PyTorch 张量: {x_torch}")  # 也变成了 99
print(f"NumPy 数组:   {x_np}")

# 如果要独立的副本，显式复制
x_torch2 = torch.tensor([1.0, 2.0, 3.0])
x_np_copy = x_torch2.numpy().copy()

# === NumPy -> PyTorch ===
np_array = np.array([[1.0, 2.0], [3.0, 4.0]])

# torch.from_numpy() - 共享内存
t_shared = torch.from_numpy(np_array)

# torch.tensor() - 复制数据
t_copy = torch.tensor(np_array)

np_array[0, 0] = 99.0
print(f"\n修改 NumPy 原数组:")
print(f"from_numpy (共享内存): \n{t_shared}")  # 跟着变了
print(f"tensor (独立副本):     \n{t_copy}")     # 不变

# === GPU 张量转 NumPy ===
if torch.cuda.is_available():
    x_gpu = torch.randn(3, 3, device="cuda")
    # GPU 张量不能直接 .numpy()，需先移到 CPU
    x_cpu = x_gpu.cpu()
    x_np = x_cpu.numpy()
    # 或一步完成
    x_np = x_gpu.cpu().numpy()
    # 推荐使用 .detach() 切断梯度后再转
    x_np = x_gpu.detach().cpu().numpy()
```

### PyTorch 与 NumPy 的主要区别

| 特性 | PyTorch 张量 | NumPy 数组 |
|------|-------------|-----------|
| GPU 支持 | 原生支持 | 不支持 |
| 自动微分 | 支持 (requires_grad) | 不支持 |
| 动态计算图 | 支持 | 不支持 |
| 数据类型 | float32 默认 | float64 默认 |
| 原地操作的命名 | 带下划线后缀 (add_) | 无统一命名规则 |
| 随机数生成 | torch.rand, torch.randn | np.random.rand, np.random.randn |
| 矩阵乘法 | @, torch.matmul | @, np.matmul, np.dot |
| 广播规则 | 与 NumPy 几乎一致 | 标准广播 |

## 本节小结

- 索引切片语法与 NumPy 高度一致，支持整数索引、切片、布尔掩码和 `...` 省略号
- `view()` 要求内存连续，`reshape()` 更安全；`permute()` 可重排多个维度
- `torch.cat()` 沿已有维度拼接，`torch.stack()` 创建新维度
- `squeeze()` 删除大小为 1 的维度，`unsqueeze()` 增加维度
- 广播规则：从后往前比较维度，兼容条件为相等或其一为 1
- 归约操作可指定 `dim` 参数沿特定维度计算，`keepdim=True` 利于后续广播
- PyTorch 张量与 NumPy 数组可通过 `.numpy()` 和 `.from_numpy()` 互相转换（注意内存共享）
