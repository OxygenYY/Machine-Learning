# 一、生成合成数据

为了让实验可控，我们先生成一个简单的线性数据集：

$$y = 3x + 5 + \varepsilon,\quad \varepsilon \sim \mathcal{N}(0, 2)$$

```python
import torch
import numpy as np
import matplotlib.pyplot as plt

# 固定随机种子，保证结果可复现
torch.manual_seed(42)
np.random.seed(42)

# 生成数据
n_samples = 200
X_np = np.random.uniform(-5, 5, size=(n_samples, 1)).astype(np.float32)
true_w = 3.0
true_b = 5.0
noise = np.random.normal(0, 2, size=(n_samples, 1)).astype(np.float32)
y_np = true_w * X_np + true_b + noise

# 转换为 PyTorch 张量
X = torch.from_numpy(X_np)
y = torch.from_numpy(y_np)

print(f"X shape: {X.shape}, y shape: {y.shape}")
print(f"X 范围: [{X.min().item():.2f}, {X.max().item():.2f}]")
print(f"y 范围: [{y.min().item():.2f}, {y.max().item():.2f}]")
```

## 划分训练集和测试集

```python
# 随机打乱索引
indices = torch.randperm(n_samples)
train_size = int(0.8 * n_samples)

train_idx = indices[:train_size]
test_idx = indices[train_size:]

X_train, y_train = X[train_idx], y[train_idx]
X_test, y_test = X[test_idx], y[test_idx]

print(f"训练集: {len(X_train)} 样本, 测试集: {len(X_test)} 样本")
```

# 二、方法一：手动梯度计算

在调用 PyTorch 的高级 API 之前，我们先手动实现一遍梯度下降，理解其内部原理。

## 模型与损失

- 模型：$\hat{y} = wx + b$
- 损失：$\mathcal{L} = \frac{1}{n}\sum (\hat{y} - y)^2$

## 梯度推导

$$\frac{\partial \mathcal{L}}{\partial w} = \frac{2}{n} \sum_{i=1}^{n} (\hat{y}^{(i)} - y^{(i)}) \cdot x^{(i)}$$

$$\frac{\partial \mathcal{L}}{\partial b} = \frac{2}{n} \sum_{i=1}^{n} (\hat{y}^{(i)} - y^{(i)})$$

```python
# 手动实现梯度下降
w = torch.tensor(0.0, requires_grad=False)  # 权重初始化为 0
b = torch.tensor(0.0, requires_grad=False)  # 偏置初始化为 0
lr = 0.01
n_epochs = 200
loss_history = []

for epoch in range(n_epochs):
    # 前向传播
    y_pred = w * X_train.squeeze() + b

    # 计算损失
    loss = torch.mean((y_pred - y_train.squeeze()) ** 2)
    loss_history.append(loss.item())

    # 手动计算梯度
    n = len(X_train)
    grad_w = (2.0 / n) * torch.sum((y_pred - y_train.squeeze()) * X_train.squeeze())
    grad_b = (2.0 / n) * torch.sum(y_pred - y_train.squeeze())

    # 梯度下降更新
    w -= lr * grad_w
    b -= lr * grad_b

    if epoch % 40 == 0:
        print(f"Epoch {epoch:3d}: Loss={loss.item():.4f}, w={w.item():.2f}, b={b.item():.2f}")

print(f"\n最终参数: w={w.item():.3f}, b={b.item():.3f}")
print(f"真实参数: w={true_w:.3f}, b={true_b:.3f}")

# 可视化的损失曲线
plt.figure(figsize=(10, 4))
plt.subplot(1, 2, 1)
plt.plot(loss_history)
plt.xlabel('Epoch')
plt.ylabel('MSE Loss')
plt.title('损失收敛曲线（手动梯度）')
plt.grid(True, alpha=0.3)
```

# 三、方法二：使用 Autograd 自动求导

PyTorch 的 `autograd` 引擎可以自动计算梯度，省去手写导数的麻烦：

```python
# 使用 autograd 实现
w_auto = torch.tensor(0.0, requires_grad=True)
b_auto = torch.tensor(0.0, requires_grad=True)
lr = 0.01
loss_history_auto = []

for epoch in range(n_epochs):
    # 前向传播
    y_pred = w_auto * X_train.squeeze() + b_auto

    # 损失
    loss = torch.mean((y_pred - y_train.squeeze()) ** 2)
    loss_history_auto.append(loss.item())

    # 反向传播：自动计算梯度
    loss.backward()

    # 更新参数（在 torch.no_grad() 中避免记录梯度）
    with torch.no_grad():
        w_auto -= lr * w_auto.grad
        b_auto -= lr * b_auto.grad

    # 梯度清零（非常关键！否则梯度会累加）
    w_auto.grad.zero_()
    b_auto.grad.zero_()

    if epoch % 40 == 0:
        print(f"Epoch {epoch:3d}: Loss={loss.item():.4f}, w={w_auto.item():.2f}, b={b_auto.item():.2f}")
```

**关键要点**：
- `loss.backward()` 从损失节点出发，沿计算图反向传播，计算所有 `requires_grad=True` 的张量的梯度。
- 更新参数时必须用 `torch.no_grad()` 包裹，防止更新操作被记录到计算图中。
- **每次迭代后必须将梯度清零** `.grad.zero_()`，否则梯度会不断累加。

# 四、方法三：使用 nn.Linear + MSELoss + SGD

这是生产中最常用的模式。`nn.Linear` 封装了权重和偏置，`nn.MSELoss` 封装了损失计算，`torch.optim.SGD` 封装了参数更新。

```python
import torch.nn as nn
import torch.optim as optim

# 定义模型
class LinearRegressionModel(nn.Module):
    def __init__(self, input_dim, output_dim):
        super().__init__()
        self.linear = nn.Linear(input_dim, output_dim)  # 包含 w 和 b

    def forward(self, x):
        return self.linear(x)

# 实例化
model = LinearRegressionModel(input_dim=1, output_dim=1)
print(f"模型结构:\n{model}")
for name, param in model.named_parameters():
    print(f"  {name}: shape={param.shape}, 初始值={param.data.item():.4f}")

# 定义损失函数和优化器
criterion = nn.MSELoss()
optimizer = optim.SGD(model.parameters(), lr=0.01)

# 训练循环
n_epochs = 200
train_losses = []

for epoch in range(n_epochs):
    # 1. 前向传播
    y_pred = model(X_train)

    # 2. 计算损失
    loss = criterion(y_pred, y_train)
    train_losses.append(loss.item())

    # 3. 反向传播
    optimizer.zero_grad()  # 清零所有参数的梯度
    loss.backward()        # 计算梯度

    # 4. 更新参数
    optimizer.step()       # 执行一步梯度下降

    if epoch % 40 == 0:
        print(f"Epoch {epoch:3d}: Loss={loss.item():.4f}")

plt.subplot(1, 2, 2)
plt.plot(loss_history_auto, label='Autograd', alpha=0.7)
plt.plot(train_losses, label='nn.Module', alpha=0.7)
plt.xlabel('Epoch')
plt.ylabel('MSE Loss')
plt.title('Autograd vs nn.Module')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.show()
```

# 五、训练循环的通用模板

无论模型多么复杂，PyTorch 训练循环的结构始终不变：

```python
def train_one_epoch(model, criterion, optimizer, X, y):
    """单轮训练的通用模板"""
    model.train()  # 训练模式（影响 Dropout、BatchNorm 等）

    # 前向传播
    predictions = model(X)
    loss = criterion(predictions, y)

    # 反向传播
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

    return loss.item()

def evaluate(model, criterion, X, y):
    """评估模式"""
    model.eval()  # 评估模式
    with torch.no_grad():  # 禁止梯度计算，节省显存
        predictions = model(X)
        loss = criterion(predictions, y)
    return loss.item(), predictions
```

**训练循环的四个步骤**：

1. **Forward**: 计算预测值
2. **Loss**: 计算误差
3. **Backward**: 计算梯度 (`loss.backward()`)
4. **Step**: 更新参数 (`optimizer.step()`)

加上梯度清零和模式切换，就构成了完整的训练模板。

# 七、评估指标

在测试集上计算 R-squared 和 MSE：

```python
from sklearn.metrics import r2_score, mean_squared_error, mean_absolute_error

model.eval()
with torch.no_grad():
    y_test_pred = model(X_test)
    y_test_pred_np = y_test_pred.numpy()
    y_test_np = y_test.numpy()

r2 = r2_score(y_test_np, y_test_pred_np)
mse = mean_squared_error(y_test_np, y_test_pred_np)
mae = mean_absolute_error(y_test_np, y_test_pred_np)
rmse = np.sqrt(mse)

print(f"{'='*40}")
print(f"测试集评估结果:")
print(f"  R²     = {r2:.4f}")
print(f"  MSE    = {mse:.4f}")
print(f"  RMSE   = {rmse:.4f}")
print(f"  MAE    = {mae:.4f}")
print(f"{'='*40}")

# 解释
if r2 > 0.9:
    print("模型解释力很强（R² > 0.9），线性假设合理。")
elif r2 > 0.7:
    print("模型具有中等解释力（0.7 < R² < 0.9）。")

    print("模型解释力较弱（R² < 0.7），数据可能包含非线性模式。")
```

# 八、完整代码汇总

以下是完整可运行的综合示例：

```python
import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
import matplotlib.pyplot as plt
from sklearn.metrics import r2_score, mean_squared_error

# ===== 1. 数据生成 =====
torch.manual_seed(42)
n_samples = 300
X = torch.linspace(-5, 5, n_samples).unsqueeze(1)
y = 3.0 * X + 5.0 + torch.randn(n_samples, 1) * 3.0

# 划分
idx = torch.randperm(n_samples)
split = int(0.8 * n_samples)
X_train, y_train = X[idx[:split]], y[idx[:split]]
X_test, y_test = X[idx[split:]], y[idx[split:]]

# ===== 2. 模型 =====
model = nn.Linear(1, 1)
criterion = nn.MSELoss()
optimizer = optim.SGD(model.parameters(), lr=0.01)

# ===== 3. 训练 =====
for epoch in range(300):
    model.train()
    y_pred = model(X_train)
    loss = criterion(y_pred, y_train)
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

# ===== 4. 评估 =====
model.eval()
with torch.no_grad():
    y_pred_test = model(X_test)
    test_loss = criterion(y_pred_test, y_test)
    r2 = r2_score(y_test.numpy(), y_pred_test.numpy())

w, b = model.weight.item(), model.bias.item()
print(f"模型: y = {w:.3f}x + {b:.3f}")
print(f"测试 MSE: {test_loss.item():.4f}, 测试 R²: {r2:.4f}")
```

从手动计算梯度，到使用 `autograd`，再到使用 `nn.Module` + `Optimizer`，我们逐步构建了工业级的线性回归训练流程。这套模板将在后续所有深度学习模型训练中反复使用。
