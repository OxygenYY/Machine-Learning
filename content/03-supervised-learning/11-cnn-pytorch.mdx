## PyTorch 卷积层 API

PyTorch 提供丰富且一致的卷积层接口。最常用的是 `nn.Conv2d`：

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import torchvision
import torchvision.transforms as T
from torch.utils.data import DataLoader
import numpy as np
import matplotlib.pyplot as plt

# nn.Conv2d 完整参数
conv = nn.Conv2d(
    in_channels=3,       # 输入通道数
    out_channels=64,     # 输出通道数（卷积核个数）
    kernel_size=3,       # 卷积核大小，可以是 int 或 (h, w)
    stride=1,            # 步长
    padding=1,           # 填充（SAME: padding = kernel_size // 2）
    dilation=1,          # 空洞率
    groups=1,            # 分组卷积（group=in_channels 时为 depthwise conv）
    bias=True,           # 是否使用偏置
    padding_mode='zeros' # 填充模式
)
print("Conv2d 参数形状:")
print("  权重 weight:", conv.weight.shape)  # (64, 3, 3, 3)
print("  偏置 bias:", conv.bias.shape)      # (64,)

# 其他常用池化层
maxpool = nn.MaxPool2d(kernel_size=2, stride=2)          # 最大池化
avgpool = nn.AvgPool2d(kernel_size=2, stride=2)          # 平均池化
adaptpool = nn.AdaptiveAvgPool2d((1, 1))                 # 自适应池化到固定尺寸

# 测试
x = torch.randn(4, 3, 32, 32)
print("\n各层输出形状:")
print("  Conv2d →", conv(x).shape)          # (4, 64, 32, 32)
print("  MaxPool →", maxpool(conv(x)).shape)  # (4, 64, 16, 16)
print("  AdaptiveAvgPool →", adaptpool(conv(x)).shape)  # (4, 64, 1, 1)
```

## 从零构建 CNN

我们构建一个经典的 CNN：3 个卷积块 + 2 个全连接层，用于 CIFAR-10 分类（32×32 彩色图像，10 类）。

```python
class SimpleCNN(nn.Module):
    """3 Conv + 2 FC 的简洁 CNN"""
    def __init__(self, num_classes=10):
        super().__init__()
        # Block 1: input 3x32x32 → output 32x16x16
        self.block1 = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.Conv2d(32, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2, 2),
            nn.Dropout2d(0.1),
        )
        # Block 2: 32x16x16 → 64x8x8
        self.block2 = nn.Sequential(
            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),
            nn.Conv2d(64, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2, 2),
            nn.Dropout2d(0.2),
        )
        # Block 3: 64x8x8 → 128x4x4
        self.block3 = nn.Sequential(
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(inplace=True),
            nn.Conv2d(128, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2, 2),
            nn.Dropout2d(0.3),
        )
        # 分类头
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(128 * 4 * 4, 256),
            nn.ReLU(inplace=True),
            nn.Dropout(0.4),
            nn.Linear(256, num_classes),
        )

    def forward(self, x):
        x = self.block1(x)
        x = self.block2(x)
        x = self.block3(x)
        x = self.classifier(x)
        return x

model = SimpleCNN()
print(f"模型参数量: {sum(p.numel() for p in model.parameters()) / 1e6:.2f}M")
```

## CIFAR-10 数据加载与增强

数据增强是提升 CNN 泛化能力的关键手段。在训练时对输入图像施加随机变换，使模型学习不变性。

```python
# 训练数据增强
train_transform = T.Compose([
    T.RandomCrop(32, padding=4),           # 先填充到 40×40，再随机裁剪回 32×32
    T.RandomHorizontalFlip(p=0.5),          # 50% 概率水平翻转
    T.ColorJitter(brightness=0.2,           # 亮度抖动
                  contrast=0.2,             # 对比度抖动
                  saturation=0.2,           # 饱和度抖动
                  hue=0.1),                 # 色调抖动
    T.ToTensor(),                           # 转为张量 [0, 1]
    T.Normalize(mean=[0.4914, 0.4822, 0.4471],  # CIFAR-10 均值
                std=[0.2470, 0.2435, 0.2616]),  # CIFAR-10 标准差
])

# 测试/验证：仅归一化
test_transform = T.Compose([
    T.ToTensor(),
    T.Normalize(mean=[0.4914, 0.4822, 0.4471],
                std=[0.2470, 0.2435, 0.2616]),
])

# 加载数据集
trainset = torchvision.datasets.CIFAR10(
    root='./data', train=True, download=True, transform=train_transform)
testset = torchvision.datasets.CIFAR10(
    root='./data', train=False, download=True, transform=test_transform)

train_loader = DataLoader(trainset, batch_size=128, shuffle=True,
                          num_workers=2, pin_memory=True)
test_loader = DataLoader(testset, batch_size=128, shuffle=False,
                         num_workers=2, pin_memory=True)

classes = ('plane', 'car', 'bird', 'cat', 'deer',
           'dog', 'frog', 'horse', 'ship', 'truck')
print(f"训练集样本: {len(trainset)}, 测试集样本: {len(testset)}")
```

## 训练循环

```python
def train_epoch(model, loader, criterion, optimizer, device):
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        running_loss += loss.item()
        _, predicted = outputs.max(1)
        total += labels.size(0)
        correct += predicted.eq(labels).sum().item()
    return running_loss / len(loader), 100. * correct / total

def evaluate(model, loader, criterion, device):
    model.eval()
    running_loss = 0.0
    correct = 0
    total = 0
    with torch.no_grad():
        for images, labels in loader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            loss = criterion(outputs, labels)
            running_loss += loss.item()
            _, predicted = outputs.max(1)
            total += labels.size(0)
            correct += predicted.eq(labels).sum().item()
    return running_loss / len(loader), 100. * correct / total

# 训练
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model = SimpleCNN().to(device)
criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=30)

num_epochs = 30
for epoch in range(num_epochs):
    train_loss, train_acc = train_epoch(model, train_loader, criterion, optimizer, device)
    test_loss, test_acc = evaluate(model, test_loader, criterion, device)
    scheduler.step()
    if (epoch + 1) % 5 == 0:
        print(f"Epoch {epoch+1:2d}/{num_epochs} | "
              f"Train Loss: {train_loss:.4f} Acc: {train_acc:.2f}% | "
              f"Test Loss: {test_loss:.4f} Acc: {test_acc:.2f}%")
```

## 可视化卷积特征图

通过注册 forward hook，我们可以提取中间层输出，观察 CNN 每一层学习到了什么特征。

```python
def visualize_feature_maps(model, image, layer_name='block1'):
    """可视化指定层的特征图"""
    activations = {}

    def hook_fn(module, input, output):
        activations['features'] = output.detach()

    # 注册 hook
    target_layer = dict(model.named_modules())[layer_name]
    handle = target_layer.register_forward_hook(hook_fn)

    model.eval()
    with torch.no_grad():
        model(image.unsqueeze(0).to(next(model.parameters()).device))
    handle.remove()

    features = activations['features'].cpu().squeeze(0)
    num_features = min(features.shape[0], 32)  # 最多显示 32 个通道

    # 绘制
    cols = 8
    rows = (num_features + cols - 1) // cols
    fig, axes = plt.subplots(rows, cols, figsize=(16, 2 * rows))
    for i in range(rows * cols):
        r, c = i // cols, i % cols
        if i < num_features:
            axes[r, c].imshow(features[i].numpy(), cmap='viridis')
        axes[r, c].axis('off')
    plt.tight_layout()
    plt.savefig(f"feature_maps_{layer_name}.png", dpi=100)
    plt.close()
    print(f"特征图可视化已保存: feature_maps_{layer_name}.png")

# 使用示例（取一个测试样本）
sample_img, sample_label = testset[0]
visualize_feature_maps(model, sample_img, 'block1')
```

## 迁移学习

迁移学习使用在大规模数据集（如 ImageNet）上预训练的模型，将其知识迁移到目标任务。这是处理小数据集最有效的策略之一。

```python
from torchvision import models

def build_transfer_model(num_classes=10, freeze_backbone=True):
    """基于 ResNet-18 的迁移学习模型"""
    # 加载预训练权重
    model = models.resnet18(pretrained=True)

    if freeze_backbone:
        # 冻结 backbone：不更新卷积层参数
        for param in model.parameters():
            param.requires_grad = False

    # 替换最后的全连接层
    in_features = model.fc.in_features
    model.fc = nn.Sequential(
        nn.Linear(in_features, 256),
        nn.ReLU(),
        nn.Dropout(0.5),
        nn.Linear(256, num_classes),
    )

    # 统计可训练参数
    trainable = sum(p.numel() for p in model.parameters() if p.requires_grad)
    total = sum(p.numel() for p in model.parameters())
    print(f"迁移学习模型: 可训练参数 {trainable:,} / 总计 {total:,}")
    return model

transfer_model = build_transfer_model(num_classes=10, freeze_backbone=True)
```

**迁移学习的常见策略**：
1. **固定特征提取器**：冻结所有卷积层，只训练新的分类头。适用于目标任务与源任务相似、数据量少的情况。
2. **微调（Fine-tuning）**：解冻部分或全部卷积层，用较小的学习率继续训练。适用于数据量中等、目标任务与源任务有一定差异的情况。

## 模型评估与混淆矩阵

```python
from sklearn.metrics import confusion_matrix, classification_report

def compute_confusion_matrix(model, loader, device):
    """计算混淆矩阵"""
    model.eval()
    all_preds = []
    all_labels = []
    with torch.no_grad():
        for images, labels in loader:
            images = images.to(device)
            outputs = model(images)
            _, preds = outputs.max(1)
            all_preds.extend(preds.cpu().tolist())
            all_labels.extend(labels.tolist())

    cm = confusion_matrix(all_labels, all_preds)
    print("\n分类报告:")
    print(classification_report(all_labels, all_preds,
                                target_names=classes, digits=3))
    return cm
```
